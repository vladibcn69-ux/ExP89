# 6.2 Verificacion SIN confiar en la CA
$ openssl verify -purpose sslserver -verify_hostname portal.proyecto.test /etc/ssl/certs/servidor.crt
CN=portal.proyecto.test
error 20 at 0 depth lookup: unable to get local issuer certificate
error /etc/ssl/certs/servidor.crt: verification failed

# 6.3 Verificacion con la CA del laboratorio
$ openssl verify -CAfile /etc/ssl/mf0489/ca.crt -purpose sslserver -verify_hostname portal.proyecto.test /etc/ssl/certs/servidor.crt
/etc/ssl/certs/servidor.crt: OK
