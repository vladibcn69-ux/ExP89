# Configuracion efectiva cargada por Nginx (nginx -T, solo bloques server)
# configuration file /etc/nginx/conf.d/default.conf:
# ExP 89 - Nginx reverse proxy con terminacion TLS
# Rutas /etc/nginx/certs/... = rutas INTERNAS del contenedor (montadas en solo lectura desde /etc/ssl del host)

server {
    listen 80;
    server_name portal.proyecto.test;

    # Redireccion permanente a HTTPS conservando host y ruta
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name portal.proyecto.test;

    ssl_certificate     /etc/nginx/certs/servidor.crt;
    ssl_certificate_key /etc/nginx/certs/servidor.key;
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_session_cache   shared:SSL:10m;
    ssl_session_timeout 10m;

    client_max_body_size 64m;

    location / {
        # "wordpress" = nombre del servicio en la red Docker exp89_front, puerto HTTP interno 80
        proxy_pass http://wordpress:80;
        proxy_http_version 1.1;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
